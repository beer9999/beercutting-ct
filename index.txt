<script>
  const API_URL = "https://script.google.com/macros/s/AKfycbwFZadcGRJK1ATdCWg6oulRAbKjOyAzyoOip_e3TNlvi5EfhZrVsx5zpX42A-9q3xLK7A/exec";

let codeReader;
let scanning = false;

async function searchPart() {

    const part = document.getElementById("part").value.trim();

    if (!part) {
        document.getElementById("result").innerHTML = "กรุณากรอก ID Part";
        return;
    }

    document.getElementById("result").innerHTML = "กำลังค้นหา...";

    try {

        const response = await fetch(API_URL + "?part=" + encodeURIComponent(part));

        const res = await response.json();

        if (res.found) {

            document.getElementById("result").innerHTML =
                "✅ พบข้อมูล<br><br>" +
                "ID Part : " + res.part + "<br>" +
                "จำนวน : " + res.qty;

        } else {

            document.getElementById("result").innerHTML = "❌ ไม่พบข้อมูล";

        }

    } catch (err) {

        document.getElementById("result").innerHTML =
            "❌ เชื่อมต่อไม่ได้<br>" + err;
    }
    
    }
async function startScanner() {

    if (scanning) return;

    scanning = true;

    try {

        codeReader = new ZXing.BrowserMultiFormatReader();

        const devices = await ZXing.BrowserCodeReader.listVideoInputDevices();

        if (!devices.length) {
            throw "ไม่พบกล้อง";
        }

        let cameraId = devices[0].deviceId;

        for (const device of devices) {

            const label = device.label.toLowerCase();

            if (
                label.includes("back") ||
                label.includes("rear") ||
                label.includes("environment")
            ) {
                cameraId = device.deviceId;
                break;
            }

        }

        await codeReader.decodeFromVideoDevice(

            cameraId,

            "video",

            (result, error) => {

                if (result) {

                    document.getElementById("part").value = result.getText();

                    codeReader.reset();

                    scanning = false;

                    searchPart();

                }

            }

        );

    } catch (err) {

        scanning = false;

        alert("เปิดกล้องไม่ได้\n\n" + err);

    }

}</script>
